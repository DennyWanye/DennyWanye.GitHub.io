在我使用新的安卓模拟器的时候，发现了这个问题：

修复的方法：

Remove the expo-dev-client dependency from your project's package.json.

Delete node_modules and run npm i.

Start fresh with npx expo start --clear.

错误4: LOG useCreateAndUpdate error: Error: Cannot write to class Novels when no flexible sync subscription has been created.

解决方法：

const ideas: any = useQueryN(ListOfRealmModels.ideas);
  const tags: any = useQueryN(ListOfRealmModels.tags);
  const collections: any = useQueryN(ListOfRealmModels.collections);
  const objects: any = useQueryN(ListOfRealmModels.object);
  const events: any = useQueryN(ListOfRealmModels.events);
  const timelines: any = useQueryN(ListOfRealmModels.timelines);
  const storys: any = useQueryN(ListOfRealmModels.storys);
  const userData: any = useQueryN(ListOfRealmModels.userData);
  //1. 添加对应的novels和chapters
  const novels: any = useQueryN(ListOfRealmModels.novels);
  const chapters: any = useQueryN(ListOfRealmModels.chapters);
  //2. 将novels和chapters加到 subscribeArray 中
  const subscribeArray: any = [ideas, tags, collections, objects, events, timelines, storys, userData, novels, chapters]

  function subscribeModel(subscribeArray: Array<any>) {
    // Create an async function so that we can `await` the
    // promise from `.subscribe()`.
    subscribeArray.forEach((item, index) => {
      const createSubscription = async () => {
        await item.subscribe({
          name: 'subscribe' + index,
          behavior: WaitForSync.FirstTime,
        });
      };
      createSubscription().catch(console.error);
    })
  }

  useEffect(() => {
    subscribeModel(subscribeArray)
  }, []);
